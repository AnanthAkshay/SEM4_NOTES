Experiment 1

🛒 Merge Sort — Stable Order ID Sorting

Online retail platform sorts thousands of orders by ID before shipping.

How it works: Split array in half → sort each half → merge both sorted halves.
Using <= during merge ensures stability (equal elements keep their original order).
ComplexityValue
TimeO(n log n)
SpaceO(n)
Stable?✅ Yes
C Implementation
MergeSort.c
#include <stdio.h>

void merge(int arr[], int l, int m, int r) {
    int n1 = m - l + 1, n2 = r - m;
    int L[n1], R[n2];

    for (int i = 0; i < n1; i++) L[i] = arr[l + i];
    for (int j = 0; j < n2; j++) R[j] = arr[m + 1 + j];

    int i = 0, j = 0, k = l;
    while (i < n1 && j < n2)
        arr[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];  // <= ensures stability
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];
}

void mergeSort(int arr[], int l, int r) {
    if (l < r) {
        int m = (l + r) / 2;
        mergeSort(arr, l, m);
        mergeSort(arr, m + 1, r);
        merge(arr, l, m, r);
    }
}

int main() {
    int orderIDs[] = {1005, 1001, 1008, 1003, 1002};
    int n = 5;
    mergeSort(orderIDs, 0, n - 1);
    printf("Sorted Order IDs: ");
    for (int i = 0; i < n; i++) printf("%d ", orderIDs[i]);
    printf("\n");
    return 0;
}
Python Implementation
MergeSort.py
def merge_sort(arr):
    if len(arr) <= 1:
        return arr

    mid = len(arr) // 2
    left  = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])

    result, i, j = [], 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:       # <= ensures stability
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result

order_ids = [1005, 1001, 1008, 1003, 1002]
print("Sorted Order IDs:", merge_sort(order_ids))
# Output: Sorted Order IDs: [1001, 1002, 1003, 1005, 1008]

📦 Randomized Quick Sort — Package Sorting

Courier service randomizes pivot selection to avoid worst-case O(n²) delays.

How it works: Pick a random pivot, partition the array (smaller left, larger right), then recursively sort both halves. Random pivot avoids sorted-input worst case.
ComplexityValue
Time (average)O(n log n)
Time (worst)O(n²) — extremely rare with random pivot
SpaceO(log n) stack
In-place?✅ Yes
C Implementation
QuickSort.c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

void swap(int *a, int *b) {
    int t = *a; *a = *b; *b = t;
}

int randomPartition(int arr[], int low, int high) {
    int r = low + rand() % (high - low + 1);
    swap(&arr[r], &arr[high]);        // move pivot to end

    int pivot = arr[high], i = low - 1;
    for (int j = low; j < high; j++)
        if (arr[j] <= pivot)
            swap(&arr[++i], &arr[j]);
    swap(&arr[++i], &arr[high]);      // place pivot in correct position
    return i;
}

void quickSort(int arr[], int low, int high) {
    if (low < high) {
        int p = randomPartition(arr, low, high);
        quickSort(arr, low, p - 1);
        quickSort(arr, p + 1, high);
    }
}

int main() {
    srand(time(0));
    int packages[] = {64, 34, 25, 12, 22, 11, 90};
    int n = 7;
    quickSort(packages, 0, n - 1);
    printf("Sorted Packages: ");
    for (int i = 0; i < n; i++) printf("%d ", packages[i]);
    printf("\n");
    return 0;
}
Python Implementation
QuickSort.py
import random

def partition(arr, low, high):
    r = random.randint(low, high)          # random pivot index
    arr[r], arr[high] = arr[high], arr[r]  # move pivot to end
    pivot = arr[high]
    i = low - 1
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i+1], arr[high] = arr[high], arr[i+1]
    return i + 1

def quick_sort(arr, low, high):
    if low < high:
        p = partition(arr, low, high)
        quick_sort(arr, low, p - 1)
        quick_sort(arr, p + 1, high)

packages = [64, 34, 25, 12, 22, 11, 90]
quick_sort(packages, 0, len(packages) - 1)
print("Sorted Packages:", packages)
# Output: Sorted Packages: [11, 12, 22, 25, 34, 64, 90]

✈️ Priority Queue (Binary Min-Heap) — Airline Boarding

Airline boards passengers by ticket class first, then by check-in time.

How it works: A min-heap keeps the highest-priority passenger at the top.
insert: add to end → bubble up. extract: remove top → bubble down.
Priority 1 = First Class (boards first), 2 = Business, 3 = Economy.
ComplexityValue
InsertO(log n)
Extract MinO(log n)
SpaceO(n)
C Implementation
PriorityQueue.c
#include <stdio.h>
#include <string.h>
#define MAX 100

typedef struct {
    int  priority;   // 1=First Class, 2=Business, 3=Economy
    int  checkin;    // Earlier check-in time = boards first (tie-break)
    char name[50];
} Passenger;

Passenger heap[MAX];
int sz = 0;

void heapSwap(int i, int j) {
    Passenger t = heap[i]; heap[i] = heap[j]; heap[j] = t;
}

int boardsBefore(int i, int j) {
    if (heap[i].priority != heap[j].priority)
        return heap[i].priority < heap[j].priority;
    return heap[i].checkin < heap[j].checkin;
}

void insert(int pri, int ci, char *name) {
    heap[sz].priority = pri;
    heap[sz].checkin  = ci;
    strcpy(heap[sz].name, name);
    int i = sz++;
    while (i > 0 && boardsBefore(i, (i-1)/2)) {
        heapSwap(i, (i-1)/2);
        i = (i-1)/2;
    }
}

Passenger extractTop() {
    Passenger top = heap[0];
    heap[0] = heap[--sz];
    int i = 0;
    while (1) {
        int l = 2*i+1, r = 2*i+2, best = i;
        if (l < sz && boardsBefore(l, best)) best = l;
        if (r < sz && boardsBefore(r, best)) best = r;
        if (best == i) break;
        heapSwap(i, best);
        i = best;
    }
    return top;
}

int main() {
    insert(3, 5,  "Alice");    // Economy
    insert(1, 10, "Bob");      // First Class
    insert(2, 3,  "Charlie");  // Business
    insert(1, 7,  "Diana");    // First Class

    char *cls[] = {"", "First Class", "Business", "Economy"};
    printf("Boarding Order:\n");
    while (sz > 0) {
        Passenger p = extractTop();
        printf("  %-10s [%s, Check-in: %d]\n", p.name, cls[p.priority], p.checkin);
    }
    return 0;
}
Python Implementation

📊 Counting Sort — Student Marks (0–100)

Exam system quickly sorts thousands of student marks, all in the range 0–100.

How it works: Count how many times each mark appears → use cumulative counts to place each element in its correct sorted position.
No comparisons needed — works only when range (k) is known and small.
ComplexityValue
TimeO(n + k) where k = 101
SpaceO(k) = O(101)
Stable?✅ Yes (right-to-left traversal)
C Implementation
CountingSort.c
#include <stdio.h>
#define MAX_MARK 101

void countingSort(int arr[], int n) {
    int count[MAX_MARK] = {0};
    int output[1000];

    for (int i = 0; i < n; i++) count[arr[i]]++;
    for (int i = 1; i < MAX_MARK; i++) count[i] += count[i-1];
    for (int i = n-1; i >= 0; i--) output[--count[arr[i]]] = arr[i];
    for (int i = 0; i < n; i++) arr[i] = output[i];
}

int main() {
    int marks[] = {85, 60, 45, 90, 72, 55, 100, 38};
    int n = 8;
    countingSort(marks, n);
    printf("Sorted Marks: ");
    for (int i = 0; i < n; i++) printf("%d ", marks[i]);
    printf("\n");
    return 0;
}
// Output: Sorted Marks: 38 45 55 60 72 85 90 100
Python Implementation
CountingSort.py
def counting_sort(arr, max_val=100):
    count = [0] * (max_val + 1)
    for x in arr:
        count[x] += 1
    result = []
    for mark in range(max_val + 1):
        result.extend([mark] * count[mark])
    return result

marks = [85, 60, 45, 90, 72, 55, 100, 38]
print("Sorted Marks:", counting_sort(marks))
# Output: Sorted Marks: [38, 45, 55, 60, 72, 85, 90, 100]

🏥 QuickSelect — Hospital Median Waiting Time

Hospital analytics finds the median patient waiting time without fully sorting.

How it works: Like Quick Sort, but after partitioning we only recurse into the side that contains the k-th element. We stop as soon as the pivot lands at position k. Expected O(n) since on average we halve the problem each time.
ComplexityValue
Time (average)O(n)
Time (worst)O(n²) — very rare with random pivot
SpaceO(1)
C Implementation
QuickSelect.c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

void swap(int *a, int *b) { int t = *a; *a = *b; *b = t; }

int partition(int arr[], int low, int high) {
    int r = low + rand() % (high - low + 1);
    swap(&arr[r], &arr[high]);
    int pivot = arr[high], i = low - 1;
    for (int j = low; j < high; j++)
        if (arr[j] <= pivot) swap(&arr[++i], &arr[j]);
    swap(&arr[++i], &arr[high]);
    return i;
}

int quickSelect(int arr[], int low, int high, int k) {
    if (low == high) return arr[low];
    int p = partition(arr, low, high);
    if (k == p)     return arr[p];
    else if (k < p) return quickSelect(arr, low, p-1, k);
    else            return quickSelect(arr, p+1, high, k);
}

int main() {
    srand(time(0));
    int waiting[] = {45, 12, 78, 34, 56, 89, 23, 67, 15, 90, 43};
    int n = 11;
    int median = quickSelect(waiting, 0, n-1, n/2);
    printf("Median waiting time: %d minutes\n", median);
    return 0;
}
// Output: Median waiting time: 45 minutes
Python Implementation
QuickSelect.py
import random

def quick_select(arr, k):
    if len(arr) == 1:
        return arr[0]
    pivot = random.choice(arr)
    low  = [x for x in arr if x < pivot]
    mid  = [x for x in arr if x == pivot]
    high = [x for x in arr if x > pivot]

    if k < len(low):
        return quick_select(low, k)
    elif k < len(low) + len(mid):
        return pivot
    else:
        return quick_select(high, k - len(low) - len(mid))

waiting_times = [45, 12, 78, 34, 56, 89, 23, 67, 15, 90, 43]
n = len(waiting_times)
median = quick_select(waiting_times, n // 2)
print("Median waiting time:", median, "minutes")
# Output: Median waiting time: 45 minutes

👥 Disjoint Set Union — Social Network Friend Groups

Social platform tracks friend groups as users form new connections.

How it works:
• Each user starts as their own group (parent = self)
• find(x): follows parent pointers to the root. Path compression: set every node's parent directly to root.
• union(x, y): find both roots; attach shorter tree under taller (union by rank).
ComplexityValue
find / unionO(α(n)) — nearly O(1)
SpaceO(n)
C Implementation
DSU.c
#include <stdio.h>
#define MAX 100

int parent[MAX], rnk[MAX];

void init(int n) {
    for (int i = 0; i < n; i++) { parent[i] = i; rnk[i] = 0; }
}

int find(int x) {
    if (parent[x] != x) parent[x] = find(parent[x]);
    return parent[x];
}

void unionSets(int x, int y) {
    int rx = find(x), ry = find(y);
    if (rx == ry) return;
    if      (rnk[rx] < rnk[ry]) parent[rx] = ry;
    else if (rnk[rx] > rnk[ry]) parent[ry] = rx;
    else { parent[ry] = rx; rnk[rx]++; }
}

int same(int x, int y) { return find(x) == find(y); }

int main() {
    init(6);
    unionSets(0, 1); unionSets(1, 2); unionSets(3, 4);
    printf("Are 0 & 2 in same group? %s\n", same(0,2) ? "Yes" : "No");
    printf("Are 0 & 3 in same group? %s\n", same(0,3) ? "Yes" : "No");
    unionSets(2, 3);
    printf("After 2 & 3 become friends:\n");
    printf("Are 0 & 4 in same group? %s\n", same(0,4) ? "Yes" : "No");
    return 0;
}
Python Implementation
DSU.py
class DSU:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank   = [0] * n

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        rx, ry = self.find(x), self.find(y)
        if rx == ry: return
        if   self.rank[rx] < self.rank[ry]: self.parent[rx] = ry
        elif self.rank[rx] > self.rank[ry]: self.parent[ry] = rx
        else: self.parent[ry] = rx; self.rank[rx] += 1

    def same(self, x, y): return self.find(x) == self.find(y)

dsu = DSU(6)
dsu.union(0, 1); dsu.union(1, 2); dsu.union(3, 4)
print("Are 0 & 2 in same group?", dsu.same(0, 2))
print("Are 0 & 3 in same group?", dsu.same(0, 3))
dsu.union(2, 3)
print("After 2 & 3 become friends:")
print("Are 0 & 4 in same group?", dsu.same(0, 4))

🗺️ Dijkstra's Shortest Path — City Navigation

City navigation system finds shortest routes between intersections.

How it works: Use a min-heap (priority queue) to always process the nearest unvisited intersection first. For each intersection, relax (update) the distances to all its neighbors.
ComplexityValue
TimeO((V + E) log V) with binary heap
SpaceO(V + E)
C Implementation
Dijkstra.c
#include <stdio.h>
#include <limits.h>
#define MAX_V 100
#define INF   INT_MAX

int adj_dest[MAX_V][MAX_V], adj_wt[MAX_V][MAX_V], adj_cnt[MAX_V];
int dist[MAX_V], visited[MAX_V];

void addEdge(int u, int v, int w) {
    int i = adj_cnt[u]++;
    adj_dest[u][i] = v; adj_wt[u][i] = w;
}

void dijkstra(int src, int V) {
    for (int i = 0; i < V; i++) { dist[i] = INF; visited[i] = 0; }
    dist[src] = 0;
    for (int iter = 0; iter < V; iter++) {
        int u = -1;
        for (int i = 0; i < V; i++)
            if (!visited[i] && (u == -1 || dist[i] < dist[u])) u = i;
        if (dist[u] == INF) break;
        visited[u] = 1;
        for (int i = 0; i < adj_cnt[u]; i++) {
            int v = adj_dest[u][i], w = adj_wt[u][i];
            if (dist[u] + w < dist[v]) dist[v] = dist[u] + w;
        }
    }
}

int main() {
    int V = 5;
    addEdge(0,1,10); addEdge(0,2,3); addEdge(1,3,2);
    addEdge(2,1,4);  addEdge(2,3,8); addEdge(3,4,5);
    dijkstra(0, V);
    printf("Shortest distances from intersection 0:\n");
    for (int i = 0; i < V; i++) printf("  To %d: %d\n", i, dist[i]);
    return 0;
}
Python Implementation
Dijkstra.py
import heapq

def dijkstra(graph, src, V):
    dist = [float('inf')] * V
    dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]: continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    return dist

V = 5
graph = {i: [] for i in range(V)}
graph[0] += [(1, 10), (2, 3)]; graph[1] += [(3, 2)]
graph[2] += [(1, 4), (3, 8)]; graph[3] += [(4, 5)]; graph[4] += []

distances = dijkstra(graph, 0, V)
print("Shortest distances from intersection 0:")
for i, d in enumerate(distances):
    print(f"  To {i}: {d}")

🎬 Matrix Chain Multiplication — Video Transformations

Video engine finds the most efficient sequence to multiply a chain of transformation matrices.

How it works: dp[i][j] = minimum scalar multiplications to compute matrix product from i to j. For each chain length, try every possible split point k and pick the minimum cost split.
ComplexityValue
TimeO(n³)
SpaceO(n²)
C Implementation
MatrixChain.c
#include <stdio.h>
#define MAX 10
#define INF 1000000000

int dp[MAX][MAX], split[MAX][MAX];

void matrixChain(int dims[], int n) {
    for (int i = 0; i < n; i++) dp[i][i] = 0;
    for (int len = 2; len <= n; len++) {
        for (int i = 0; i <= n - len; i++) {
            int j = i + len - 1;
            dp[i][j] = INF;
            for (int k = i; k < j; k++) {
                int cost = dp[i][k] + dp[k+1][j] + dims[i]*dims[k+1]*dims[j+1];
                if (cost < dp[i][j]) { dp[i][j] = cost; split[i][j] = k; }
            }
        }
    }
}

void printOrder(int i, int j) {
    if (i == j) { printf("M%d", i+1); return; }
    printf("("); printOrder(i, split[i][j]);
    printf(" x "); printOrder(split[i][j]+1, j); printf(")");
}

int main() {
    int dims[] = {30, 35, 15, 5, 10};
    int n = 4;
    matrixChain(dims, n);
    printf("Min scalar multiplications: %d\n", dp[0][n-1]);
    printf("Optimal order: "); printOrder(0, n-1); printf("\n");
    return 0;
}
Python Implementation
MatrixChain.py
def matrix_chain(dims):
    n = len(dims) - 1
    dp    = [[0]*n for _ in range(n)]
    split = [[0]*n for _ in range(n)]
    for length in range(2, n+1):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i][j] = float('inf')
            for k in range(i, j):
                cost = dp[i][k] + dp[k+1][j] + dims[i]*dims[k+1]*dims[j+1]
                if cost < dp[i][j]:
                    dp[i][j] = cost; split[i][j] = k
    return dp, split

def print_order(split, i, j):
    if i == j: return f"M{i+1}"
    k = split[i][j]
    return f"({print_order(split, i, k)} x {print_order(split, k+1, j)})"

dims = [30, 35, 15, 5, 10]
dp, split = matrix_chain(dims)
n = len(dims) - 1
print("Min scalar multiplications:", dp[0][n-1])
print("Optimal order:", print_order(split, 0, n-1))
Experiment 9

📄 LCS — Plagiarism Detection

Plagiarism detection system finds the longest matching character sequence between two texts.

How it works: Build a 2D table. dp[i][j] = length of LCS of first i chars of s1 and first j chars of s2. If characters match: diagonal + 1. Else: max of left or top. Traceback the table to recover the actual LCS string.
ComplexityValue
TimeO(m × n)
SpaceO(m × n)
C Implementation
LCS.c
#include <stdio.h>
#include <string.h>
#define MAX 100

int dp[MAX][MAX];

void lcs(char *s1, char *s2) {
    int m = strlen(s1), n = strlen(s2);
    for (int i = 0; i <= m; i++) dp[i][0] = 0;
    for (int j = 0; j <= n; j++) dp[0][j] = 0;

    for (int i = 1; i <= m; i++)
        for (int j = 1; j <= n; j++)
            if (s1[i-1] == s2[j-1]) dp[i][j] = dp[i-1][j-1] + 1;
            else dp[i][j] = (dp[i-1][j] > dp[i][j-1]) ? dp[i-1][j] : dp[i][j-1];

    // Traceback
    int len = dp[m][n];
    char result[MAX]; result[len] = '\0';
    int i = m, j = n;
    while (i > 0 && j > 0) {
        if (s1[i-1] == s2[j-1]) { result[--len] = s1[i-1]; i--; j--; }
        else if (dp[i-1][j] > dp[i][j-1]) i--;
        else j--;
    }
    printf("LCS Length : %d\n", dp[m][n]);
    printf("LCS String : %s\n", result);
}

int main() {
    lcs("ALGORITHM", "ALTRUISTIC");
    return 0;
}
Python Implementation
LCS.py
def lcs(s1, s2):
    m, n = len(s1), len(s2)
    dp = [[0] * (n+1) for _ in range(m+1)]
    for i in range(1, m+1):
        for j in range(1, n+1):
            if s1[i-1] == s2[j-1]: dp[i][j] = dp[i-1][j-1] + 1
            else: dp[i][j] = max(dp[i-1][j], dp[i][j-1])

    result, i, j = [], m, n
    while i > 0 and j > 0:
        if s1[i-1] == s2[j-1]:
            result.append(s1[i-1]); i -= 1; j -= 1
        elif dp[i-1][j] > dp[i][j-1]: i -= 1
        else: j -= 1
    lcs_str = ''.join(reversed(result))
    print(f"LCS Length : {dp[m][n]}")
    print(f"LCS String : {lcs_str}")

📡 Huffman Coding — Text Compression

Telecom provider compresses text messages by assigning shorter codes to frequent characters.

How it works: Build a min-heap of characters by frequency. Repeatedly: pop two lowest-frequency nodes → merge them into a parent node (sum of frequencies) → push back. Final tree gives prefix codes: frequent chars get shorter bit strings.
ComplexityValue
TimeO(n log n)
SpaceO(n)
StrategyGreedy ✅
C Implementation
Huffman.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define ALPHA 256

typedef struct Node {
    char ch; int freq;
    struct Node *left, *right;
} Node;

Node* newNode(char ch, int freq) {
    Node *n = (Node*)malloc(sizeof(Node));
    n->ch = ch; n->freq = freq; n->left = n->right = NULL;
    return n;
}

Node *heap[ALPHA]; int hsz = 0;

void push(Node *n) {
    heap[hsz++] = n;
    int i = hsz - 1;
    while (i > 0 && heap[(i-1)/2]->freq > heap[i]->freq) {
        Node *t = heap[(i-1)/2]; heap[(i-1)/2] = heap[i]; heap[i] = t;
        i = (i-1)/2;
    }
}

Node* pop() {
    Node *top = heap[0]; heap[0] = heap[--hsz];
    int i = 0;
    while (1) {
        int l = 2*i+1, r = 2*i+2, s = i;
        if (l < hsz && heap[l]->freq < heap[s]->freq) s = l;
        if (r < hsz && heap[r]->freq < heap[s]->freq) s = r;
        if (s == i) break;
        Node *t = heap[i]; heap[i] = heap[s]; heap[s] = t; i = s;
    }
    return top;
}

void printCodes(Node *root, char *code, int depth) {
    if (!root->left && !root->right) {
        code[depth] = '\0';
        printf("  '%c' (freq=%2d) : %s\n", root->ch, root->freq, code);
        return;
    }
    if (root->left)  { code[depth] = '0'; printCodes(root->left,  code, depth+1); }
    if (root->right) { code[depth] = '1'; printCodes(root->right, code, depth+1); }
}

int main() {
    char text[] = "telecom";
    int freq[ALPHA] = {0};
    for (int i = 0; text[i]; i++) freq[(unsigned char)text[i]]++;
    for (int i = 0; i < ALPHA; i++) if (freq[i]) push(newNode((char)i, freq[i]));
    while (hsz > 1) {
        Node *l = pop(), *r = pop();
        Node *par = newNode('\0', l->freq + r->freq);
        par->left = l; par->right = r; push(par);
    }
    char code[ALPHA];
    printf("Huffman Codes for \"%s\":\n", text);
    printCodes(heap[0], code, 0);
    return 0;
}
Python Implementation
Huffman.py
import heapq

class Node:
    def __init__(self, ch, freq):
        self.ch, self.freq = ch, freq
        self.left = self.right = None
    def __lt__(self, other): return self.freq < other.freq

def huffman(text):
    freq = {}
    for c in text: freq[c] = freq.get(c, 0) + 1
    heap = [Node(c, f) for c, f in freq.items()]
    heapq.heapify(heap)
    while len(heap) > 1:
        left, right = heapq.heappop(heap), heapq.heappop(heap)
        parent = Node('\0', left.freq + right.freq)
        parent.left, parent.right = left, right
        heapq.heappush(heap, parent)
    codes = {}
    def gen_codes(node, code=""):
        if node:
            if not node.left and not node.right:
                codes[node.ch] = code or "0"
            gen_codes(node.left, code + "0")
            gen_codes(node.right, code + "1")
    gen_codes(heap[0])
    print(f'Huffman Codes for "{text}":')
    for ch in sorted(codes):
        print(f"  '{ch}' (freq={freq[ch]:2d}) : {codes[ch]}")

💧 Maximum Flow — Water Supply Network

City water supply distributes from a reservoir to multiple zones through pipelines with limited capacity.

How it works (Ford-Fulkerson with BFS = Edmonds-Karp):
1. Find a path from source to sink using BFS
2. Find the bottleneck (minimum capacity edge on the path)
3. Push that much flow through; subtract from forward edges, add to reverse edges
4. Repeat until no path from source to sink exists
ComplexityValue
TimeO(V × E²)
SpaceO(V²)
C Implementation
MaxFlow.c
#include <stdio.h>
#include <string.h>
#define MAX 10
#define INF 1000000000

int cap[MAX][MAX], parent[MAX], visited[MAX], q[MAX*MAX];

int bfs(int s, int t, int n) {
    memset(visited, 0, sizeof(visited));
    int front = 0, rear = 0;
    q[rear++] = s; visited[s] = 1; parent[s] = -1;
    while (front < rear) {
        int u = q[front++];
        for (int v = 0; v < n; v++) {
            if (!visited[v] && cap[u][v] > 0) {
                visited[v] = 1; parent[v] = u;
                if (v == t) return 1;
                q[rear++] = v;
            }
        }
    }
    return 0;
}

int maxFlow(int s, int t, int n) {
    int total = 0;
    while (bfs(s, t, n)) {
        int flow = INF;
        for (int v = t; v != s; v = parent[v])
            if (cap[parent[v]][v] < flow) flow = cap[parent[v]][v];
        for (int v = t; v != s; v = parent[v]) {
            cap[parent[v]][v] -= flow;
            cap[v][parent[v]] += flow;
        }
        total += flow;
    }
    return total;
}

int main() {
    int n = 6;
    memset(cap, 0, sizeof(cap));
    cap[0][1]=16; cap[0][2]=13; cap[1][2]=10; cap[1][3]=12;
    cap[2][4]=14; cap[3][5]=20; cap[3][2]=9;  cap[4][3]=7; cap[4][5]=4;
    printf("Maximum water flow: %d units\n", maxFlow(0, 5, n));
    return 0;
}
// Output: Maximum water flow: 23 units
Python Implementation
MaxFlow.py
from collections import deque

def bfs(cap, s, t, parent, n):
    visited = [False] * n
    visited[s] = True
    queue = deque([s])
    while queue:
        u = queue.popleft()
        for v in range(n):
            if not visited[v] and cap[u][v] > 0:
                visited[v] = True; parent[v] = u
                if v == t: return True
                queue.append(v)
    return False

def max_flow(cap, s, t, n):
    parent, total_flow = [-1]*n, 0
    while bfs(cap, s, t, parent, n):
        path_flow, v = float('inf'), t
        while v != s:
            u = parent[v]; path_flow = min(path_flow, cap[u][v]); v = u
        v = t
        while v != s:
            u = parent[v]; cap[u][v] -= path_flow; cap[v][u] += path_flow; v = u
        total_flow += path_flow
    return total_flow

cap = [[0]*n for _ in range(n)]
cap[0][1]=16; cap[0][2]=13; cap[1][2]=10; cap[1][3]=12
cap[2][4]=14; cap[3][5]=20; cap[3][2]=9;  cap[4][3]=7; cap[4][5]=4
print("Maximum water flow:", max_flow(cap, 0, 5, n), "units")

🗣️ Viva Voce — Top 50 Questions & Answers

Comprehensive oral examination preparation containing high-frequency questions for the Design and Analysis of Algorithms laboratory course.

1. What is an algorithm?
An algorithm is a step-by-step procedure or a set of rules to solve a specific computational problem in a finite amount of time.
2. What are the criteria for a good algorithm?
A good algorithm must satisfy five conditions: Input (zero or more values), Output (at least one result), Definiteness (clear, unambiguous steps), Finiteness (must terminate after a finite number of steps), and Effectiveness (each step must be basic enough to be carried out).
3. What is asymptotic notation? Name the primary types.
Asymptotic notation is a mathematical framework to describe the limiting execution behavior of an algorithm as the input size grows. The primary types are:
• Big-O (O): Represents the asymptotic upper bound (worst-case).
• Big-Omega (Ω): Represents the asymptotic lower bound (best-case).
• Big-Theta (Θ): Represents the asymptotic tight bound (average-case).
4. What is the difference between Big-O and Big-Theta notation?
Big-O notation represents an asymptotic upper bound only (e.g., f(n) = O(g(n)) means f(n) grows no faster than g(n)). Big-Theta represents a tight bound (e.g., f(n) = Θ(g(n)) means f(n) is bounded both above and below by g(n) asymptotically, i.e., it has the same rate of growth).
5. What is the divide-and-conquer strategy?
It is an algorithmic design paradigm that solves a problem by:
1. Divide: Splitting the problem into smaller, independent subproblems of the same type.
2. Conquer: Solving the subproblems recursively (or directly if they are base cases).
3. Combine: Merging the subproblem solutions to construct the solution to the original problem.
6. Explain the recurrence relation for Merge Sort.
The recurrence relation for Merge Sort is: T(n) = 2T(n/2) + Θ(n).
Here, 2T(n/2) represents the recursive sorting of two halves, and Θ(n) represents the time taken to merge the two sorted halves. Solving this recurrence using the Master Theorem yields T(n) = O(n log n).
7. Why is Merge Sort stable, and what does stability mean?
A sorting algorithm is stable if it preserves the relative order of duplicate elements. Merge Sort is stable because during the merge step, when comparing elements from the left subarray (L) and right subarray (R), we select from L if the elements are equal (using L[i] <= R[j]). This prevents equal elements from swapping positions.
8. What is the auxiliary space complexity of Merge Sort, and why?
The auxiliary space complexity is O(n) because Merge Sort requires temporary left and right arrays to store elements during the merge phase before copying them back to the original array.
9. How does Quick Sort work, and what is its average-case time complexity?
Quick Sort works by selecting a "pivot" element, partitioning the array such that elements smaller than the pivot go to the left and larger elements go to the right, and then recursively sorting the left and right partitions. Its average-case time complexity is O(n log n).
10. What is the worst-case time complexity of Quick Sort, and how does randomized partitioning help?
The worst-case complexity is O(n²), which occurs when the pivot consistently divides the array into extremely unbalanced partitions (e.g., size 0 and n-1), typically when the input is already sorted or reverse-sorted. Randomized Quick Sort solves this by choosing a random pivot at each step, making the worst-case highly improbable and ensuring an expected runtime of O(n log n).
11. Compare Merge Sort and Quick Sort.
• Space: Merge Sort uses O(n) auxiliary space; Quick Sort runs in-place (uses only recursive stack space, O(log n) average).
• Stability: Merge Sort is stable; Quick Sort is unstable.
• Performance: Quick Sort has a smaller constant factor and is generally faster in-place on RAM; Merge Sort is preferred for linked lists or external sorting.
12. What is a binary heap, and what are its properties?
A binary heap is a complete binary tree (all levels filled except possibly the last, which is filled left-to-right) that satisfies the heap property:
• Min-Heap Property: The value of each node is greater than or equal to the value of its parent (root is minimum).
• Max-Heap Property: The value of each node is less than or equal to the value of its parent (root is maximum).
13. How does a binary heap represent itself in an array?
Since it is a complete binary tree, it can be stored compactly in a 1D array without pointers. For any element at index i (0-indexed):
• Left Child index = 2 * i + 1
• Right Child index = 2 * i + 2
• Parent index = (i - 1) / 2 (integer division)
14. What are the time complexities of basic Min-Heap operations?
• Insert: O(log n) (requires bubble-up / sift-up).
• Extract-Min: O(log n) (replaces root with last element and runs sift-down / heapify).
• Get-Min: O(1) (accesses index 0).
• Build-Heap: O(n) (bottom-up heapification of all non-leaf nodes).
15. What is Counting Sort, and when is it preferred?
Counting Sort is a non-comparison-based sorting algorithm that works by counting the occurrences of each unique element. It is preferred when the input consists of integers within a relatively small, known range k (i.e., k = O(n)), where it achieves linear time complexity.
16. Why is Counting Sort stable, and what is its time complexity?
Counting Sort is stable because it iterates through the input array backwards in the placement phase, utilizing cumulative counts to determine position and placing equal elements in the same relative order. Its time complexity is O(n + k), where n is the size of the array and k is the range of key values.
17. What is QuickSelect, and what is its main application?
QuickSelect is a selection algorithm designed to find the k-th smallest element in an unsorted array (e.g., finding the median). It uses the same partitioning logic as Quick Sort but only recurses into one partition containing the target rank instead of both.
18. What are the average and worst-case time complexities of QuickSelect?
The average-case complexity is O(n) because the search space is cut in half at each step (sum of n + n/2 + n/4 ... = 2n). The worst-case is O(n²), which occurs when partitions are highly unbalanced (e.g., pivot is consistently the minimum or maximum element).
19. What is the Disjoint Set Union (DSU) data structure?
DSU (or Union-Find) is a data structure that tracks elements partitioned into disjoint sets. It supports two primary operations:
• Find(x): Determines which set element x belongs to (returns the representative/root of the set).
• Union(x, y): Merges the sets containing x and y.
20. Explain union-by-rank and path compression in DSU.
• Union-by-Rank: Attaches the root of the tree with smaller height (rank) to the root of the tree with larger height, keeping the tree flat.
• Path Compression: Directs all nodes traversed during a Find call directly to the root, dramatically flattening the tree for subsequent operations.
21. What is the time complexity of DSU operations with rank and path compression?
Amortized complexity per operation is O(α(n)), where α is the inverse Ackermann function. This function grows so slowly that α(n) <= 4 for all practical inputs, making DSU operations virtually constant time, O(1).
22. What is the Greedy choice property?
The Greedy choice property states that a globally optimal solution can be reached by making a locally optimal (greedy) choice at each step without ever revising previous decisions.
23. What is Dijkstra's algorithm used for, and what are its constraints?
Dijkstra's algorithm finds the shortest paths from a single source vertex to all other vertices in a weighted graph. Its critical constraint is that all edge weights must be non-negative.
24. Why doesn't Dijkstra's algorithm work with negative edge weights?
Dijkstra's algorithm is greedy and marks a vertex's shortest distance as finalized ("visited") once it is extracted from the queue. If negative weight edges are present, a path discovered later could bypass the finalized distance and yield a smaller cost, which Dijkstra fails to correct.
25. What is the time complexity of Dijkstra's algorithm using a min-heap?
The time complexity is O((V + E) log V), where V is the number of vertices and E is the number of edges, because each vertex is inserted/extracted from the heap, and edge relaxations trigger heap updates.
26. What is Dynamic Programming (DP), and how does it differ from Divide & Conquer?
Dynamic Programming solves optimization problems by breaking them into overlapping subproblems, solving each subproblem exactly once, and caching their results (memoization or tabulation) to avoid redundant calculations. In contrast, Divide & Conquer works on independent (non-overlapping) subproblems, which results in repeated computations if subproblems overlap.
27. What are the key properties a problem must have to be solved by Dynamic Programming?
1. Optimal Substructure: The optimal solution to the main problem contains optimal solutions to its subproblems.
2. Overlapping Subproblems: The recursive solution solves the same subproblems repeatedly instead of generating new ones.
28. Explain the Matrix Chain Multiplication (MCM) problem.
MCM is an optimization problem to find the most efficient parenthesization to multiply a chain of matrices. It does not perform the actual multiplication but computes the ordering of multiplications that minimizes the total scalar multiplication operations.
29. What is the recurrence relation and time complexity of Matrix Chain DP?
The recurrence is: m[i, j] = min_{i <= k < j} (m[i, k] + m[k+1, j] + p_{i-1} * p_k * p_j) where m[i, j] is the minimum cost to multiply matrices A_i through A_j. The time complexity is O(n³) and space complexity is O(n²).
30. What is the Longest Common Subsequence (LCS) problem?
LCS finds the longest subsequence common to two sequences, where the elements do not need to occupy consecutive positions in the original strings (unlike substrings). It is commonly used in difference detection utilities (like diff) and bioinformatics.
31. What is the time and space complexity of the LCS dynamic programming algorithm?
For strings of lengths m and n, the dynamic programming solution takes O(m * n) time and O(m * n) space to store the DP table.
32. What is Huffman Coding, and is it a greedy or DP algorithm?
Huffman Coding is a greedy algorithm used for lossless data compression. It assigns variable-length binary codes to characters based on their frequencies; more frequent characters get shorter codes, minimizing the total encoded file size.
33. What is the prefix rule in Huffman Coding?
The prefix rule states that no codeword can be a prefix of another codeword. This ensures that the compressed binary stream can be parsed and decoded uniquely and unambiguously from left to right without needing separation markers.
34. What is the time complexity to build a Huffman tree?
The complexity is O(n log n), where n is the number of unique characters. This is because we repeatedly extract two minimum elements from a priority queue (min-heap) and insert a combined node n-1 times.
35. What is Depth First Search (DFS), and how are discovery and finishing times useful?
DFS is a graph traversal algorithm that explores as deep as possible along each branch before backtracking. Timestamps:
• Discovery Time d[u]: Count when node u is first visited.
• Finishing Time f[u]: Count when all descendants of u have been fully explored.
These times help classify edges, find topological orders, detect cycles, and identify strongly connected components.
36. Explain tree edges, back edges, forward edges, and cross edges in DFS.
• Tree Edge: Edge traversed to discover a new vertex.
• Back Edge: Edge pointing from a vertex to an ancestor in the DFS tree (indicates a cycle).
• Forward Edge: Non-tree edge pointing from a vertex to a descendant in the DFS tree.
• Cross Edge: Any other edge between branches or tree components with no ancestor relationship.
37. What is the Maximum Flow problem, and what are its physical constraints?
The Max Flow problem seeks to find the maximum volume of flow that can go from a source to a sink through a capacity-constrained network. It must satisfy:
1. Capacity Constraint: Flow on any edge cannot exceed its capacity (f(u,v) <= c(u,v)).
2. Flow Conservation: Net flow into a node equals net flow out of a node for all vertices except source and sink.
38. Explain the Ford-Fulkerson method and Edmonds-Karp algorithm.
Ford-Fulkerson is a general method that iteratively finds augmenting paths in a residual network, pushes bottleneck flow along these paths, and updates residual capacities until no augmenting paths exist. Edmonds-Karp is a specific implementation of Ford-Fulkerson that uses BFS to find the shortest augmenting path (in terms of number of edges), guaranteeing termination.
39. What is the time complexity of the Edmonds-Karp algorithm?
The time complexity is O(V * E²), which is independent of the maximum flow values.
40. What is a residual graph in flow networks?
A residual graph G_f represents the remaining capacities of edges in a flow network. It contains:
• Forward edges: indicating how much more flow can be sent (c(u,v) - f(u,v)).
• Backward edges: indicating how much flow can be redirected or cancelled in the reverse direction (f(u,v)).
41. What is the Master Theorem, and when is it used?
The Master Theorem is a recipe for solving recurrence relations of the form T(n) = aT(n/b) + f(n), which commonly arise in divide-and-conquer algorithms where a >= 1 is the number of subproblems, b > 1 is the subproblem reduction factor, and f(n) is the divide/combine cost.
42. Explain the three cases of the Master Theorem.
Let k = log_b(a):
• Case 1: If f(n) = O(n^(k - ε)) for some ε > 0, then T(n) = Θ(n^k).
• Case 2: If f(n) = Θ(n^k), then T(n) = Θ(n^k * log n).
• Case 3: If f(n) = Ω(n^(k + ε)) for some ε > 0, and satisfies the regularity condition a*f(n/b) <= c*f(n), then T(n) = Θ(f(n)).
43. What is NP-Complete, and how does it differ from NP-Hard?
• NP (Nondeterministic Polynomial-time): Problems whose solutions can be verified in polynomial time.
• NP-Hard: Problems to which any problem in NP can be reduced in polynomial time (at least as hard as any NP problem).
• NP-Complete: Problems that are both in NP and are NP-Hard. They represent the hardest problems in NP.
44. What is the difference between recursion and iteration?
Recursion solves a problem by having a function call itself with smaller inputs (storing execution states in the system call stack, consuming O(depth) memory). Iteration uses loop structures (like for/while) to repeat instructions, using a constant amount of memory, O(1).
45. What is memoization in dynamic programming?
Memoization is the top-down approach of dynamic programming. It uses recursion to solve the problem and saves the output of each subproblem in a lookup table (e.g., hash map or array) so that the next time the subproblem is encountered, the cached result is returned directly.
46. What is tabulation in dynamic programming?
Tabulation is the bottom-up approach of dynamic programming. It solves base cases first, then iteratively fills a table (usually a 1D or 2D array) from the bottom up to compute the solution for larger values, avoiding recursive call overhead.
47. What is a topological sort, and which graphs support it?
A topological sort is a linear ordering of vertices such that for every directed edge u -> v, vertex u comes before v. Only Directed Acyclic Graphs (DAGs) support topological sorting.
48. How can DFS be used to perform topological sorting?
By running DFS on the DAG. When a node finished (all its neighbors are visited and it is about to backtrack), we push it onto a stack. After DFS finishes, the elements popped from the stack one by one represent the topological order.
49. What is space complexity, and how is auxiliary space defined?
Space complexity is the total memory space required by an algorithm during execution, including the memory for input data. Auxiliary space is the temporary or extra space used by the algorithm, excluding the input data.
50. What is an in-place sorting algorithm? Give examples.
An in-place sorting algorithm rearranges the input array elements directly within the original array structure, requiring only a constant or logarithmic amount of auxiliary memory (O(1) or O(log n)) for execution. Examples: Quick Sort, Heap Sort, Bubble Sort, Insertion Sort.
